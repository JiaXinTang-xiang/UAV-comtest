cmake_minimum_required(VERSION 3.16)
project(bao_vm_gui LANGUAGES CXX)
find_package(gazebo REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(bao_vm_gui SHARED vm_gui.cpp)
target_compile_features(bao_vm_gui PRIVATE cxx_std_17)
target_include_directories(bao_vm_gui SYSTEM PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(bao_vm_gui PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(bao_vm_gui PRIVATE ${GAZEBO_LIBRARIES} gazebo_gui Qt5::Widgets)
